   Tabelas:
   CLIENTES (id_cliente, nome, cidade, categoria)
   PEDIDOS  (id_pedido, id_cliente, valor_total, quantidade_itens, dias_entrega)
   



-- QUESTAO 1 - Analise de vendas por cidade
-- Para cada cidade: qtd de pedidos, faturamento total, valor medio
-- e o menor valor de pedido. So entram cidades com >= 5 pedidos.
-- =================================================================
SELECT 
    c.cidade,
    COUNT(p.id_pedido) AS qtd_pedidos,
    SUM(p.valor_total) AS faturamento_total,
    AVG(p.valor_total) AS valor_medio_pedido,
    MIN(p.valor_total) AS menor_valor_pedido
FROM CLIENTES c
INNER JOIN PEDIDOS p ON c.id_cliente = p.id_cliente
GROUP BY c.cidade
HAVING COUNT(p.id_pedido) >= 5;



-- QUESTAO 2 - Categorias com resultado comercial relevante
-- Uma categoria so aparece se, ao mesmo tempo:
--   i)   tiver pelo menos 5 pedidos
--   ii)  faturamento total > R$ 5.000,00
--   iii) valor medio dos pedidos > R$ 500,00

select 
    c.categoria,
    COUNT(p.id_pedido) AS qtd_pedidos,
    SUM(p.valor_total) AS faturamento_total,
    AVG(p.valor_total) AS valor_medio_pedido,
    MIN(p.valor_total) AS menor_valor_pedido
FROM CLIENTES c
INNER JOIN PEDIDOS p ON c.id_cliente = p.id_cliente
GROUP BY c.categoria
HAVING COUNT(p.id_pedido) >= 5
   AND SUM(p.valor_total) > 5000
   AND AVG(p.valor_total) > 500;



-- QUESTAO 3 - Clientes com relacionamento comercial significativo
-- So aparecem clientes que, ao mesmo tempo:
--   i)   fizeram pelo menos 3 pedidos
--   ii)  gastaram mais de R$ 2.000,00 no total
--   iii) tem valor medio de pedido > R$ 700,00
-- Agrupo por id_cliente junto com o nome para nao misturar dois
-- clientes diferentes que, por coincidencia, tenham o mesmo nome.

SELECT 
    c.nome,
    COUNT(p.id_pedido) AS qtd_pedidos,
    SUM(p.valor_total) AS total_gasto,
    AVG(p.valor_total) AS valor_medio_pedido,
    MIN(p.valor_total) AS menor_pedido
FROM CLIENTES c
INNER JOIN PEDIDOS p ON c.id_cliente = p.id_cliente
GROUP BY c.id_cliente, c.nome
HAVING COUNT(p.id_pedido) >= 3
   AND SUM(p.valor_total) > 2000
   AND AVG(p.valor_total) > 700;



-- QUESTAO 4 - Analise de vendas e logistica por cidade
-- Relaciona volume de vendas com desempenho de entrega.
-- So entram cidades que, ao mesmo tempo:
--   i)   tem pelo menos 5 pedidos
--   ii)  venderam mais de 20 itens
--   iii) tem media de entrega menor que 5 dias
-- Todas as condicoes usando HAVING, como pedido no enunciado.

SELECT 
    c.cidade,
    COUNT(p.id_pedido) AS qtd_pedidos,
    SUM(p.quantidade_itens) AS qtd_itens_vendidos,
    AVG(p.dias_entrega) AS media_dias_entrega,
    MIN(p.dias_entrega) AS menor_prazo_entrega
FROM CLIENTES c
INNER JOIN PEDIDOS p ON c.id_cliente = p.id_cliente
GROUP BY c.cidade
HAVING COUNT(p.id_pedido) >= 5
   AND SUM(p.quantidade_itens) > 20
   AND AVG(p.dias_entrega) < 5;



-- QUESTAO 5 - Analise simultanea por cidade e categoria
-- Agora agrupando por DUAS colunas ao mesmo tempo (cidade e
-- categoria). So aparecem combinacoes com pelo menos 3 pedidos.

SELECT 
    c.cidade,
    c.categoria,
    COUNT(p.id_pedido) AS qtd_pedidos,
    SUM(p.valor_total) AS faturamento_total,
    AVG(p.valor_total) AS valor_medio_pedido,
    MIN(p.valor_total) AS menor_valor_pedido
FROM CLIENTES c
INNER JOIN PEDIDOS p ON c.id_cliente = p.id_cliente
GROUP BY c.cidade, c.categoria
HAVING COUNT(p.id_pedido) >= 3;



-- QUESTAO 6 - Desafio: segmentos comercialmente fortes
-- Uma combinacao cidade/categoria e "forte" quando, ao mesmo tempo:
--   i)   tem pelo menos 3 pedidos
--   ii)  faturamento total >= R$ 2.500,00
--   iii) valor medio dos pedidos >= R$ 700,00
--   iv)  nenhum pedido do grupo vale menos que R$ 600,00
--
-- O item (iv) e o pulo do gato da questao: nao existe uma funcao
-- pronta de "nenhum valor abaixo de X", mas se o MENOR valor do
-- grupo (MIN) ja e >= 600, entao OBRIGATORIAMENTE todos os outros
-- valores do grupo tambem sao >= 600 (ninguem pode ser menor que
-- o proprio minimo). Por isso basta filtrar MIN(valor_total) >= 600.

SELECT 
    c.cidade,
    c.categoria,
    COUNT(p.id_pedido) AS qtd_pedidos,
    SUM(p.valor_total) AS faturamento_total,
    AVG(p.valor_total) AS valor_medio_pedido,
    MIN(p.valor_total) AS menor_valor_pedido
FROM CLIENTES c
INNER JOIN PEDIDOS p ON c.id_cliente = p.id_cliente
GROUP BY c.cidade, c.categoria
HAVING COUNT(p.id_pedido) >= 3
   AND SUM(p.valor_total) >= 2500
   AND AVG(p.valor_total) >= 700
   AND MIN(p.valor_total) >= 600;



   QUESTAO 7 - Interpretacao dos resultados de agregacao


a) Por que uma condicao baseada em COUNT deve ser colocada em
   HAVING e nao em WHERE?

   Porque o WHERE e executado ANTES do GROUP BY, filtrando as
   linhas originais da tabela uma por uma (linha de pedido por
   linha de pedido). Nesse momento os grupos ainda nem existem,
   entao nao tem como calcular um COUNT, que so faz sentido depois
   que os dados ja foram agrupados. Ja o HAVING roda DEPOIS do
   GROUP BY, entao ele filtra os grupos que ja foram formados. Por
   isso qualquer condicao com funcao de agregacao (COUNT, SUM, AVG,
   MIN) tem que ficar no HAVING.


b) Qual e a diferenca entre agrupar somente por cidade e agrupar
   por cidade, categoria?

   Ao agrupar so por cidade, todos os pedidos daquela cidade caem
   num unico grupo, misturando clientes de categorias diferentes
   (Ouro, Prata, Bronze) numa linha de resultado so. Ao agrupar por
   cidade E categoria, e criado um grupo para cada combinacao
   diferente das duas colunas - ou seja, dentro da mesma cidade,
   os pedidos de cada categoria ficam separados em linhas
   diferentes. Isso da uma visao mais detalhada (granular) do
   negocio, pois mostra o desempenho de cada categoria dentro de
   cada cidade, e nao so o total geral da cidade.


c) Por que MIN(valor_total) pode ser utilizado para verificar se
   todos os pedidos de um grupo possuem valor de pelo menos
   R$ 600,00?

   Porque o MIN(valor_total) retorna exatamente o MENOR valor
   dentre todos os pedidos daquele grupo. Se ate esse menor valor
   for maior ou igual a 600, isso ja garante automaticamente que
   nenhum outro pedido do grupo pode ser menor que 600 (por
   definicao, nao existe valor menor que o proprio minimo).
   Entao checar "MIN(valor_total) >= 600" e equivalente a checar
   "todos os pedidos do grupo sao >= 600", sem precisar olhar
   pedido por pedido.


d) Considere um grupo com pedidos de R$ 500,00, R$ 900,00,
   R$ 1.000,00 e R$ 1.200,00. Esse grupo pode satisfazer
   AVG(valor_total) >= 700? Esse mesmo grupo pode satisfazer
   MIN(valor_total) >= 600? Justifique.

   AVG = (500 + 900 + 1000 + 1200) / 4 = 3600 / 4 = 900
   Como 900 >= 700, SIM, esse grupo satisfaz a condicao de media.

   MIN = 500 (o menor valor entre os quatro pedidos)
   Como 500 < 600, esse grupo NAO satisfaz a condicao do minimo.
